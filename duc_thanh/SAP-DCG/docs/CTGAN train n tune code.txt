# Core CTGAN training

import numpy as np
import torch
from ctgan import CTGAN


def train_ctgan(
    df,
    discrete_columns,
    epochs=300,
    batch_size=500,
    pac=10,
    seed=42,
):
    torch.manual_seed(seed)
    np.random.seed(seed)

    model = CTGAN(
        epochs=epochs,
        batch_size=batch_size,
        pac=pac,
        verbose=True,
        enable_gpu=torch.cuda.is_available(),
    )

    model.fit(df, discrete_columns=discrete_columns)
    return model

# Optuna hyperparameter tuning

import time
import optuna

DIMS = {
    "256x256": (256, 256),
    "256x256x256": (256, 256, 256),
    "512x512": (512, 512),
}


def sample_params(trial):
    return {
        "epochs": trial.suggest_int("epochs", 10, 50, step=10),
        "batch_size": trial.suggest_categorical(
            "batch_size", [500, 1000]
        ),
        "pac": trial.suggest_categorical(
            "pac", [1, 5, 10]
        ),
        "embedding_dim": trial.suggest_categorical(
            "embedding_dim", [64, 128, 256]
        ),
        "generator_dim": DIMS[
            trial.suggest_categorical(
                "generator_dim", list(DIMS)
            )
        ],
        "discriminator_dim": DIMS[
            trial.suggest_categorical(
                "discriminator_dim", list(DIMS)
            )
        ],
        "generator_lr": trial.suggest_float(
            "generator_lr", 1e-5, 1e-3, log=True
        ),
        "discriminator_lr": trial.suggest_float(
            "discriminator_lr", 1e-5, 1e-3, log=True
        ),
        "discriminator_steps": trial.suggest_categorical(
            "discriminator_steps", [1, 3, 5]
        ),
        "log_frequency": trial.suggest_categorical(
            "log_frequency", [True, False]
        ),
    }
